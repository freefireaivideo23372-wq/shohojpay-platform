package com.shohojpay.smsdetector

import android.content.BroadcastReceiver
import android.content.Context
import android.content.Intent
import android.provider.Telephony
import org.json.JSONObject
import java.io.OutputStreamWriter
import java.net.HttpURLConnection
import java.net.URL
import kotlin.concurrent.thread

class SmsBroadcastReceiver : BroadcastReceiver() {
    private val MFS_SENDERS = listOf("bkash", "nagad", "16216", "upay", "16247", "16167")

    override fun onReceive(context: Context, intent: Intent) {
        if (intent.action != Telephony.Sms.Intents.SMS_RECEIVED_ACTION) return
        val messages = Telephony.Sms.Intents.getMessagesFromIntent(intent) ?: return
        val sender = messages[0].originatingAddress ?: ""
        val body = messages.joinToString("") { it.messageBody }

        // শুধুমাত্র পেমেন্ট মেসেজ ফিল্টার করা হচ্ছে (ব্যক্তিগত SMS সম্পূর্ণ নিরাপদ)
        if (MFS_SENDERS.any { sender.contains(it, true) } || body.contains("TrxID", true) || body.contains("TxnID", true)) {
            val prefs = context.getSharedPreferences("ShohojPayPrefs", Context.MODE_PRIVATE)
            val webhookUrl = prefs.getString("webhook_url", "https://yourdomain.com/api.php?action=sms_webhook") ?: return
            val apiKey = prefs.getString("api_key", "sp_live_98a72b41c65d89e01f23a45b") ?: return

            thread {
                try {
                    val conn = (URL(webhookUrl).openConnection() as HttpURLConnection).apply {
                        requestMethod = "POST"
                        setRequestProperty("Content-Type", "application/json")
                        setRequestProperty("shohojpay-api-key", apiKey)
                        doOutput = true
                    }
                    val payload = JSONObject().apply {
                        put("sender", sender)
                        put("message", body)
                        put("received_at", System.currentTimeMillis())
                    }
                    OutputStreamWriter(conn.outputStream).use { it.write(payload.toString()) }
                    conn.responseCode
                    conn.disconnect()
                } catch (e: Exception) { e.printStackTrace() }
            }
        }
    }
}
